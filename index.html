Cycle Fleet Management

Tracks the physical fleet — including wear. Every trip ages a cycle, and after enough trips it automatically pulls itself out of service until an admin runs maintenance.

Encapsulation Static counters (fleet + service queue) operator++ (prefix)
┌───────────── CampusCycle ─────────────┐ │ - cycleId : int │ │ - model[30], dock[30] : char │ │ - inUse, needsMaintenance : bool │ │ - tripCount : int │ wear counter │ - totalCycles, retiredForService: static│ ├────────────────────────────────────── ┤ │ + CampusCycle() / (id,model,dock) │ │ + CampusCycle(const CampusCycle&) │ │ + checkOut() / checkIn(newDock) │ │ + service() → clears wear, re-enters │ │ + operator++() → ages the cycle │ ← mutates + can retire the object └────────────────────────────────────── ┘
operator++ controls the cycle's own lifecycle
CampusCycle& operator++() {
  tripCount++;
  if (tripCount >= MAINTENANCE_THRESHOLD && !needsMaintenance) {
    needsMaintenance = true;      // pulled from service automatically
    retiredForService++;
  }
  return *this;
}

This is the overload the assignment's own wording points at — using an operator "to control object creation, destruction, and operator behaviour." Here ++cycle isn't decoration: it's the one place a cycle's fate is decided.

/*
    Smart Campus Bike Sharing Management System
    Course: DSA01 - Object Oriented Programming with C++
    ------------------------------------------------------------------
    Design angle: instead of a plain rental ledger, the system tracks
    the sustainability impact of every trip (SDG 7 / 11 / 12 from the
    brief) — eco-points, rider loyalty tiers, peak-hour pricing, and
    cycle wear/maintenance — so the class design and operator use are
    driven by that theme rather than a generic "rent-a-bike" template.

    OOP coverage:
      - Classes, encapsulation, access specifiers, member functions
      - Default / parameterized / copy constructors + destructors
      - Static members (campus-wide counters)
      - Arrays of objects
      - Operator overloading used for actual system behaviour, not just
        display: prefix ++ ages a cycle toward maintenance, += folds a
        trip's eco-points into a rider, and > ranks trips by CO2 saved.
*/

#include <iostream>
#include <iomanip>
#include <cstring>
using namespace std;

const int MAX_RIDERS = 50;
const int MAX_CYCLES = 50;
const int MAX_LOGS   = 100;
const int MAINTENANCE_THRESHOLD = 15;   // trips before a cycle needs servicing
const float CO2_PER_KM_GRAMS   = 120.0f; // avg car emissions avoided per km cycled

// ---------------------------------------------------------
// MODULE 1 : RIDER MANAGEMENT -> class Rider
// ---------------------------------------------------------
class Rider {
private:
    int riderId;
    char name[40];
    char category[10];     // "Student" or "Staff"
    int ecoPoints;          // earned from trips, drives the loyalty tier
    static int totalRiders;

    const char* computeTier() const {
        if (ecoPoints >= 150) return "Gold";
        if (ecoPoints >= 50)  return "Silver";
        return "Bronze";
    }

public:
    Rider() {
        riderId = 0; strcpy(name, "N/A"); strcpy(category, "N/A"); ecoPoints = 0;
    }

    Rider(int id, const char* n, const char* cat) {
        riderId = id; strcpy(name, n); strcpy(category, cat); ecoPoints = 0;
        totalRiders++;
    }

    Rider(const Rider &r) {
        riderId = r.riderId; strcpy(name, r.name); strcpy(category, r.category);
        ecoPoints = r.ecoPoints; totalRiders++;
    }

    ~Rider() { /* rider leaves the scheme; object lifetime ends cleanly */ }

    int getId() const { return riderId; }
    int getEcoPoints() const { return ecoPoints; }
    float discountForTier() const {
        const char* t = computeTier();
        if (strcmp(t, "Gold") == 0)   return 0.15f;
        if (strcmp(t, "Silver") == 0) return 0.05f;
        return 0.0f;
    }

    // Operator overload #1: folds a completed trip's eco-points straight
    // into the rider (defined properly once TripLog is declared below).
    Rider& operator+=(int pointsEarned) {
        ecoPoints += pointsEarned;
        return *this;
    }

    void display() const {
        cout << left << setw(6) << riderId << setw(18) << name << setw(9) << category
             << setw(8) << ecoPoints << computeTier() << endl;
    }

    static int getTotalRiders() { return totalRiders; }
};
int Rider::totalRiders = 0;

// ---------------------------------------------------------
// MODULE 2 : CYCLE FLEET MANAGEMENT -> class CampusCycle
// ---------------------------------------------------------
class CampusCycle {
private:
    int cycleId;
    char model[30];
    char dock[30];
    bool inUse;
    int tripCount;          // wear counter
    bool needsMaintenance;
    static int totalCycles;
    static int retiredForService;

public:
    CampusCycle() {
        cycleId = 0; strcpy(model, "N/A"); strcpy(dock, "Dock-0");
        inUse = false; tripCount = 0; needsMaintenance = false;
    }

    CampusCycle(int id, const char* m, const char* dockName) {
        cycleId = id; strcpy(model, m); strcpy(dock, dockName);
        inUse = false; tripCount = 0; needsMaintenance = false;
        totalCycles++;
    }

    CampusCycle(const CampusCycle &c) {
        cycleId = c.cycleId; strcpy(model, c.model); strcpy(dock, c.dock);
        inUse = c.inUse; tripCount = c.tripCount; needsMaintenance = c.needsMaintenance;
        totalCycles++;
    }

    ~CampusCycle() { /* cycle decommissioned / object destroyed */ }

    int getId() const { return cycleId; }
    bool isAvailable() const { return !inUse && !needsMaintenance; }
    bool flaggedForService() const { return needsMaintenance; }

    void checkOut() { inUse = true; }
    void checkIn(const char* newDock) { inUse = false; strcpy(dock, newDock); }

    void service() {                 // admin action: clears wear, re-enters fleet
        if (needsMaintenance) { needsMaintenance = false; tripCount = 0; retiredForService--; }
    }

    // Operator overload #2: prefix ++ represents "one more trip taken" on
    // this exact cycle. It mutates object state and, once the wear
    // threshold is crossed, automatically pulls the cycle out of service —
    // an operator directly controlling object behaviour, as the brief asks.
    CampusCycle& operator++() {
        tripCount++;
        if (tripCount >= MAINTENANCE_THRESHOLD && !needsMaintenance) {
            needsMaintenance = true;
            retiredForService++;
        }
        return *this;
    }

    void display() const {
        cout << left << setw(6) << cycleId << setw(14) << model << setw(11) << dock
             << setw(6) << tripCount
             << (needsMaintenance ? "Needs service" : (inUse ? "In-Use" : "Available")) << endl;
    }

    static int getTotalCycles() { return totalCycles; }
    static int getRetiredForService() { return retiredForService; }
};
int CampusCycle::totalCycles = 0;
int CampusCycle::retiredForService = 0;

// ---------------------------------------------------------
// MODULE 3 : TRIP MANAGEMENT -> class TripLog
// ---------------------------------------------------------
class TripLog {
private:
    int logId;
    int riderId;
    int cycleId;
    float distanceKm;
    float minutesUsed;
    bool peakHour;
    float fare;
    int ecoPointsEarned;
    float co2SavedGrams;
    static int totalTrips;
    static float totalFareCollected;
    static float totalCO2SavedGrams;

    float computeFare(float discount) const {
        const float BASE = 5.0f, PER_KM = 3.0f, PER_MIN = 0.4f;
        float raw = BASE + (distanceKm * PER_KM) + (minutesUsed * PER_MIN);
        if (peakHour) raw *= 1.2f;           // peak-hour surcharge
        return raw * (1.0f - discount);      // loyalty discount applied last
    }

public:
    TripLog() {
        logId = 0; riderId = 0; cycleId = 0; distanceKm = 0; minutesUsed = 0;
        peakHour = false; fare = 0; ecoPointsEarned = 0; co2SavedGrams = 0;
    }

    TripLog(int id, int rId, int cId, float dist, float mins, bool peak, float riderDiscount) {
        logId = id; riderId = rId; cycleId = cId;
        distanceKm = dist; minutesUsed = mins; peakHour = peak;
        fare = computeFare(riderDiscount);
        // off-peak trips earn double points to nudge demand away from rush hour
        ecoPointsEarned = (int)(distanceKm * (peak ? 1 : 2));
        co2SavedGrams = distanceKm * CO2_PER_KM_GRAMS;
        totalTrips++;
        totalFareCollected += fare;
        totalCO2SavedGrams += co2SavedGrams;
    }

    TripLog(const TripLog &t) {
        logId = t.logId; riderId = t.riderId; cycleId = t.cycleId;
        distanceKm = t.distanceKm; minutesUsed = t.minutesUsed; peakHour = t.peakHour;
        fare = t.fare; ecoPointsEarned = t.ecoPointsEarned; co2SavedGrams = t.co2SavedGrams;
        totalTrips++; totalFareCollected += fare; totalCO2SavedGrams += co2SavedGrams;
    }

    ~TripLog() { /* trip closed out; cycle already checked back in */ }

    int getLogId() const { return logId; }
    int getRiderId() const { return riderId; }
    int getCycleId() const { return cycleId; }
    float getFare() const { return fare; }
    int getEcoPoints() const { return ecoPointsEarned; }
    float getCO2Saved() const { return co2SavedGrams; }

    // Operator overload #3: '>' ranks two trips by environmental impact
    // (CO2 saved) rather than by cost — a comparison a plain rental
    // system would never need, but this system's theme calls for.
    bool operator>(const TripLog &other) const {
        return this->co2SavedGrams > other.co2SavedGrams;
    }

    friend ostream& operator<<(ostream &out, const TripLog &t) {
        out << "Trip#" << t.logId << " | Rider " << t.riderId << " | Cycle " << t.cycleId
            << " | " << t.distanceKm << " km | Rs." << fixed << setprecision(2) << t.fare
            << " | +" << t.ecoPointsEarned << " pts | " << (int)t.co2SavedGrams << "g CO2 saved";
        return out;
    }

    static int getTotalTrips() { return totalTrips; }
    static float getTotalFareCollected() { return totalFareCollected; }
    static float getTotalCO2SavedGrams() { return totalCO2SavedGrams; }
};
int TripLog::totalTrips = 0;
float TripLog::totalFareCollected = 0;
float TripLog::totalCO2SavedGrams = 0;

// ---------------------------------------------------------
// Arrays of objects (bounded roster / fleet / trip history)
// ---------------------------------------------------------
Rider riders[MAX_RIDERS];
CampusCycle cycles[MAX_CYCLES];
TripLog logs[MAX_LOGS];
int riderCount = 0, cycleCount = 0, logCount = 0;

int findRider(int id) { for (int i = 0; i < riderCount; i++) if (riders[i].getId() == id) return i; return -1; }
int findCycle(int id)  { for (int i = 0; i < cycleCount; i++) if (cycles[i].getId() == id) return i; return -1; }

void registerRider() {
    if (riderCount >= MAX_RIDERS) { cout << "Roster full.\n"; return; }
    int id; char name[40], cat[10];
    cout << "Rider ID: "; cin >> id;
    cout << "Name: "; cin >> name;
    cout << "Category (Student/Staff): "; cin >> cat;
    riders[riderCount++] = Rider(id, name, cat);
    cout << "Rider registered.\n";
}

void listRiders() {
    cout << "\n-- Riders (Total: " << Rider::getTotalRiders() << ") --\n";
    cout << left << setw(6) << "ID" << setw(18) << "Name" << setw(9) << "Type"
         << setw(8) << "Points" << "Tier" << endl;
    for (int i = 0; i < riderCount; i++) riders[i].display();
}

void riderMenu() {
    int c;
    do {
        cout << "\n=== MODULE 1: RIDER MANAGEMENT ===\n1. Register Rider\n2. View Riders\n0. Back\nChoice: ";
        cin >> c;
        if (c == 1) registerRider(); else if (c == 2) listRiders();
    } while (c != 0);
}

// MODULE 2 MENU : Cycle Fleet Management

void addCycle() {
    if (cycleCount >= MAX_CYCLES) { cout << "Fleet full.\n"; return; }
    int id; char model[30], dock[30];
    cout << "Cycle ID: "; cin >> id;
    cout << "Model: "; cin >> model;
    cout << "Dock: "; cin >> dock;
    cycles[cycleCount++] = CampusCycle(id, model, dock);
    cout << "Cycle added to fleet.\n";
}

void viewFleet() {
    cout << "\n-- Fleet (Total: " << CampusCycle::getTotalCycles()
         << " | Needs service: " << CampusCycle::getRetiredForService() << ") --\n";
    cout << left << setw(6) << "ID" << setw(14) << "Model" << setw(11) << "Dock"
         << setw(6) << "Trips" << "Status" << endl;
    for (int i = 0; i < cycleCount; i++) cycles[i].display();
}

void serviceCycle() {
    int id; cout << "Cycle ID to service: "; cin >> id;
    int ci = findCycle(id);
    if (ci == -1) { cout << "Not found.\n"; return; }
    cycles[ci].service();
    cout << "Cycle serviced and back in the fleet.\n";
}

void cycleMenu() {
    int c;

        cin >> c;
        if (c == 1) addCycle(); else if (c == 2) viewFleet(); else if (c == 3) serviceCycle();
    } while (c != 0);
}

// ---------------------------------------------------------
// MODULE 3 MENU : Trip Management
// ---------------------------------------------------------
void logTrip() {
    if (logCount >= MAX_LOGS) { cout << "Trip log full.\n"; return; }
    int rid, cid; float dist, mins; char peakAns;
    cout << "Rider ID: "; cin >> rid;
    cout << "Cycle ID: "; cin >> cid;
    int ri = findRider(rid), ci = findCycle(cid);
    if (ri == -1 || ci == -1) { cout << "Invalid Rider/Cycle ID.\n"; return; }
    if (!cycles[ci].isAvailable()) { cout << "Cycle unavailable (in use or needs service).\n"; return; }

    cycles[ci].checkOut();
    cout << "Distance (km), Duration (min): "; cin >> dist >> mins;
    cout << "Is this a peak hour trip? (y/n): "; cin >> peakAns;
    bool peak = (peakAns == 'y' || peakAns == 'Y');

    logs[logCount] = TripLog(logCount + 1, rid, cid, dist, mins, peak, riders[ri].discountForTier());
    cout << "Trip logged -> " << logs[logCount] << endl;

    riders[ri] += logs[logCount].getEcoPoints();   // operator+= folds points into the rider
    ++cycles[ci];                                   // operator++ ages the cycle toward service
    cycles[ci].checkIn("Nearest Dock");
    logCount++;
}

void listTrips() {
    cout << "\n-- Trip Log (Total: " << TripLog::getTotalTrips() << ") --\n";
    for (int i = 0; i < logCount; i++) cout << logs[i] << endl;
}

void tripMenu() {
    int c;

        cout << "\n=== MODULE 3: TRIP MANAGEMENT ===\n1. Log a Trip\n2. View Trip Log\n0. Back\nChoice: ";
        cin >> c;
        if (c == 1) logTrip(); else if (c == 2) listTrips();
    } while (c != 0);
}

// ---------------------------------------------------------
// MODULE 4 MENU : Impact Reports & Comparison
// ---------------------------------------------------------
void compareTrips() {
    if (logCount < 2) { cout << "Need at least 2 trips.\n"; return; }
    int a, b;
    cout << "First trip ID: "; cin >> a;
    cout << "Second trip ID: "; cin >> b;
    if (a < 1 || a > logCount || b < 1 || b > logCount) { cout << "Invalid ID.\n"; return; }
    TripLog &t1 = logs[a - 1], &t2 = logs[b - 1];
    cout << t1 << "\n" << t2 << "\n";
    if (t1 > t2) cout << "Trip #" << a << " was the greener ride.\n";        // operator>
    else if (t2 > t1) cout << "Trip #" << b << " was the greener ride.\n";
    else cout << "Both trips saved the same CO2.\n";
}

void impactReport() {
    cout << "\n=== CAMPUS IMPACT REPORT ===\n";
    cout << "Total Riders           : " << Rider::getTotalRiders() << endl;
    cout << "Total Cycles in Fleet  : " << CampusCycle::getTotalCycles()
         << " (needing service: " << CampusCycle::getRetiredForService() << ")\n";
    cout << "Total Trips Logged     : " << TripLog::getTotalTrips() << endl;
    cout << "Total Fare Collected   : Rs." << fixed << setprecision(2)
         << TripLog::getTotalFareCollected() << endl;
    cout << "Total CO2 Avoided      : " << fixed << setprecision(0)
         << TripLog::getTotalCO2SavedGrams() / 1000.0f << " kg" << endl;

    int available = 0;
    for (int i = 0; i < cycleCount; i++) if (cycles[i].isAvailable()) available++;
    cout << "Cycles Currently Free  : " << available << " / " << cycleCount << endl;
}

void reportMenu() {
    int c;
    do {
        cout << "\n=== MODULE 4: IMPACT REPORTS & COMPARISON ===\n1. Compare Two Trips (greener ride)\n2. Campus Impact Report\n0. Back\nChoice: ";
        cin >> c;
        if (c == 1) compareTrips(); else if (c == 2) impactReport();
    } while (c != 0);
}
